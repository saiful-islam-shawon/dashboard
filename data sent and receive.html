/** @odoo-module **/

import { registry } from "@web/core/registry";
import { Component, useState, onWillStart } from "@odoo/owl";
import { useService } from "@web/core/utils/hooks";

export class Dashboard extends Component {
    setup() {

        this.state = useState({
            products: 0
        });

        onWillStart(async () => {
            // api
            await this.get_products_from_api()
        });
    }

    async get_products_from_api() {
        const result = await fetch('/api/products/', {
            method: 'POST',
            headers: {
                'Content-Type': 'application/json',
            },
            body: JSON.stringify({
                jsonrpc: '2.0',
                method: 'call',
                params: {}
            })
        });
        
        const data = await result.json();
        this.state.products = data.result.products;
        console.log(this.state.products)
    }
}

Dashboard.template = "practice_module.custom_dashboard_view";

registry.category("actions").add(
    "custom_dashboard",
    Dashboard
);

from odoo import http
from odoo.http import request
import json

class ProductApiController(http.Controller):

@http.route('/api/products', auth='user', type='json', methods=['POST'])
def get_products(self):
    products = request.env['product.template'].sudo().search_read(
        [],
        ['id', 'name', 'qty_available', 'list_price']
    )
    return {'products': products, 'total': len(products)}
