step 1: create model file

from odoo import models, fields

class SalesInventoryDashboard(models.Model):
  _name = 'custom.kpi.dashboard'
  _description = 'Custom KPI dashboard for sales and inventory'
  
  total_sales = fields.Float(string='Total Sales')
  total_order = fields.Integer(string='Total Orders')
  total_stock = fields.Float(string='Total Stock')
  
  def action_refresh_dashboard(self):
      for record in self:
          record.total_sales = sum(self.env["sale.order"].search([("state", "=", "sale")]).mapped("amount_total"))
          record.total_order = self.env["sale.order"].search_count([("state", "=", "sale")])
          
          
          products = self.env["product.product"].search([])
          record.total_stock = sum(products.mapped("qty_available"))
