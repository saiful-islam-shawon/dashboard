step 3: static/src/components/sales_dashboard.js

/** @odoo-module **/

const { DateTime } = luxon;
import { registry } from "@web/core/registry";
import { Component, useState, onWillStart } from "@odoo/owl";

import { KpiCard } from "./kpi_card/kpi_card";
import { ChartRender } from "./chart_render/chart_render";

import {useService} from "@web/core/utils/hooks"

export class OwlSalesDashboard extends Component {
    setup(){
        this.state = useState({
            quotation: {
                value:10,
                percentage:20
            },
            period:90,
            date:null
        })

        this.orm = useService("orm")

        onWillStart(async ()=>{
            this.getDate()
            await this.getQuotation()
        })
    }

    async onChangePeriod(){
        
        this.getDate()
        await this.getQuotation()
    }

    getDate(){
        const date = DateTime.now()
        .minus({ days: this.state.period })
        .toUTC()
        .toFormat("yyyy-MM-dd HH:mm:ss");

        this.state.date = date;
    }

    async getQuotation(){
        const data = await this.orm.searchCount("sale.order", [['state', 'in', ['sent', 'draft']], ['create_date', '>', this.state.date]])
        this.state.quotation.value = data 
    }
}

OwlSalesDashboard.template =
    "practice_module.OwlSalesDashboard";

OwlSalesDashboard.components = {
    KpiCard,
    ChartRender,
};

registry.category("actions").add(
    "sales_dashboard",
    OwlSalesDashboard
);

/** @odoo-module **/

import {
    Component,
    onWillStart,
    useRef,
    onMounted,
} from "@odoo/owl";

import { loadJS } from "@web/core/assets";

export class ChartRender extends Component {

    setup() {

        this.chartRef = useRef("chart");

        onWillStart(async () => {

            await loadJS(
                "https://cdnjs.cloudflare.com/ajax/libs/Chart.js/4.5.0/chart.umd.min.js"
            );

        });

        onMounted(() => {

            const data = [
                { year: 2010, count: 10 },
                { year: 2011, count: 20 },
                { year: 2012, count: 15 },
                { year: 2013, count: 25 },
                { year: 2014, count: 22 },
                { year: 2015, count: 30 },
                { year: 2016, count: 28 },
                { year: 2017, count: 28 },
            ];

            new Chart(this.chartRef.el, {
                type: "doughnut",
                data: {
                    labels: data.map(row => row.year),
                    datasets: [
                        {
                            label: "Acquisitions by year",
                            data: data.map(row => row.count),
                            backgroundColor: [
                                "#ca0303",
                                "#f05811",
                                "#ffce56",
                                "#4bc0c0",
                                "#9966ff",
                                "#ff9f40",
                                "#66ff66",
                                "#000000"
                            ]
                        }
                    ]
                }
            });

        });

    }
}

ChartRender.template =
    "practice_module.ChartRender";
