/** @odoo-module **/

import { registry } from "@web/core/registry";
import { Component, useState, onWillStart } from "@odoo/owl";
import { useService } from "@web/core/utils/hooks";
import { Learn } from "./chart_dashboard/chart_dashboard";

export class Dashboard extends Component {
    setup() {

        this.state = useState({
            total_quotation: 0,
            total_order: 0,
            total_product:0,
            total_purchase_rfq:0,
            total_purchase_order:0
        });

        this.orm = useService("orm");

        onWillStart(async () => {
            await this.getQuotation();
            await this.getOrders()
            await this.get_total_product()
            await this.get_total_purchase_rfq()
            await this.get_total_purchase_order()
        });
    }

    async getQuotation() {

        const data = await this.orm.searchCount(
            "sale.order",
            [['state', 'in', ['sent', 'draft']]]
        );

        this.state.total_quotation = data;
        console.log(this.state.total_quotation)
    }

    async getOrders(){
        const data = await this.orm.searchCount("sale.order", [["state", '=', 'sale']])
        this.state.total_order = data
    }

    async get_total_product(){
        const data = await this.orm.searchCount("product.template", [['qty_available', '>', 0]])
        this.state.total_product = data
    }

    async get_total_purchase_rfq(){
        const data = await this.orm.searchCount("purchase.order", [['state', 'in', ['draft', 'sent']]]);
        this.state.total_purchase_rfq = data
    }

    async get_total_purchase_order(){
        const data = await this.orm.searchCount("purchase.order", [["state", "in", ["purchase", "done"]]])
        this.state.total_purchase_order = data
    }
}

Dashboard.template = "practice_module.custom_dashboard_view";
Dashboard.components = {
    Learn
}

registry.category("actions").add(
    "custom_dashboard",
    Dashboard
);

/** @odoo-module **/

// OWL framework থেকে প্রয়োজনীয় functions import করা হয়েছে
import {
    Component,      // Base class for all OWL components
    onWillStart,    // Component mount হওয়ার আগে async কাজ করার জন্য
    useRef,         // DOM element reference নেওয়ার জন্য
    onMounted,      // Component mount হওয়ার পরে কাজ করার জন্য
    useState        // Reactive state manage করার জন্য
} from "@odoo/owl";

import { useService } from "@web/core/utils/hooks"; // Odoo services (orm, rpc) use করার জন্য
import { loadJS } from "@web/core/assets";          // External JS file load করার জন্য

export class Learn extends Component {

    setup() {
        // ORM service inject করা হয়েছে, database query করার জন্য
        this.orm = useService("orm");

        // Canvas element এর reference নেওয়া হয়েছে, Chart.js এটা ব্যবহার করবে
        this.chartRef = useRef("chart");

        // Reactive state declare করা হয়েছে
        // এই values change হলে UI automatically update হবে
        this.state = useState({
            quotation: 0,   // Total quotation count
            sale_order: 0,  // Total sale order count
        });

        // Component render হওয়ার আগে এই block execute হবে
        onWillStart(async () => {

            // Chart.js library load করা হয়েছে CDN থেকে
            await loadJS(
                "https://cdnjs.cloudflare.com/ajax/libs/Chart.js/4.5.0/chart.umd.min.js"
            );

            // Chart এর উপরে percentage দেখানোর জন্য datalabels plugin load করা হয়েছে
            await loadJS(
                "https://cdn.jsdelivr.net/npm/chartjs-plugin-datalabels@2.2.0/dist/chartjs-plugin-datalabels.min.js"
            );

            // Database থেকে sale data fetch করা হয়েছে
            await this.getSaleData();
        });

        // Component DOM এ mount হওয়ার পরে chart render করা হবে
        // onWillStart এর পরে execute হয়, তাই data আগেই ready থাকে
        onMounted(() => {
            this.renderChart();
        });
    }

    // Sale order data database থেকে fetch করার method
    async getSaleData() {

        // Draft এবং Sent state এর sale order গুলো count করা হয়েছে (Quotations)
        const quotationCount = await this.orm.searchCount(
            "sale.order",
            [["state", "in", ["draft", "sent"]]]
        );

        // Sale এবং Done state এর sale order গুলো count করা হয়েছে (Confirmed Orders)
        const saleCount = await this.orm.searchCount(
            "sale.order",
            [["state", "in", ["sale", "done"]]]
        );

        // Fetched data state এ save করা হয়েছে
        this.state.quotation = quotationCount;
        this.state.sale_order = saleCount;
    }

    // Chart render করার method
    renderChart() {
        new Chart(this.chartRef.el, {

            // Doughnut type chart ব্যবহার করা হয়েছে
            type: "doughnut",

            data: {
                // Chart এর labels
                labels: ["Quotations", "Sale Orders"],

                datasets: [
                    {
                        label: "Sales Overview",

                        // State থেকে real data ব্যবহার করা হয়েছে
                        data: [this.state.quotation, this.state.sale_order],

                        // প্রতিটা slice এর color
                        // Quotation = Orange, Sale Order = Green
                        backgroundColor: [
                            "#f05811",  // Orange - Quotations
                            "#11998e",  // Green  - Sale Orders
                        ]
                    }
                ]
            },

            options: {
                responsive: true,           // Container size অনুযায়ী chart resize হবে
                maintainAspectRatio: false, // Parent div এর height অনুযায়ী chart fit হবে

                plugins: {
                    // Chart এর উপরে legend দেখানোর জন্য
                    legend: {
                        position: "top"
                    },

                    // Hover করলে tooltip এ percentage দেখাবে
                    tooltip: {
                        callbacks: {
                            label: function(context) {
                                // Total value calculate করা হয়েছে
                                const total = context.dataset.data.reduce((a, b) => a + b, 0);
                                const value = context.parsed;

                                // Percentage calculate করা হয়েছে
                                // total 0 হলে 0% দেখাবে (divide by zero এড়ানোর জন্য)
                                const percentage = total > 0 ? ((value / total) * 100).toFixed(1) : 0;

                                // Example: "Quotations: 5 (62.5%)"
                                return `${context.label}: ${value} (${percentage}%)`;
                            }
                        }
                    },

                    // Chart এর slice এর উপরে সবসময় percentage দেখানোর জন্য
                    datalabels: {
                        color: "#fff",      // সাদা রঙে দেখাবে
                        font: {
                            size: 16,       // Font size
                            weight: "bold"  // Bold text
                        },
                        formatter: function(value, context) {
                            // Total value calculate করা হয়েছে
                            const total = context.dataset.data.reduce((a, b) => a + b, 0);

                            // Percentage calculate করা হয়েছে (1 decimal point)
                            const percentage = total > 0 ? ((value / total) * 100).toFixed(1) : 0;

                            // Example: "62.5%"
                            return percentage + "%";
                        }
                    }
                }
            },

            // ChartDataLabels plugin globally register করা হয়েছে
            // এটা ছাড়া datalabels কাজ করবে না
            plugins: [ChartDataLabels]
        });
    }
}

// এই component এর XML template এর নাম
Learn.template = "practice_module.ChartUse";
